A Python function that says whether a word is a palindrome. Code only.

def is_palindrome(word: str) -> bool:
    """Return True if the given word is a palindrome, ignoring case and non-alphanumeric characters."""
    cleaned = ''.join(c.lower() for c in word if c.isalnum())
    return cleaned == cleaned[::-1]
